## Media Queries in CSS Modules

### Basic Media Query Usage

```css
/* Card.module.css */
.card {
  padding: 16px;
  display: grid;
  grid-template-columns: 1fr;
  gap: 16px;
}

@media (min-width: 768px) {
  .card {
    padding: 24px;
    grid-template-columns: 1fr 1fr;
  }
}

@media (min-width: 1024px) {
  .card {
    padding: 32px;
    grid-template-columns: repeat(3, 1fr);
  }
}
```

### Using CSS Variables for Breakpoints

```css
/* variables.css */
:root {
  --breakpoint-sm: 640px;
  --breakpoint-md: 768px;
  --breakpoint-lg: 1024px;
  --breakpoint-xl: 1280px;
}
```

<Callout type="warn" title="CSS Variable Limitation">
CSS variables cannot be used directly in media query definitions. Use a preprocessor or define breakpoints as constants in both CSS and JS.
</Callout>

### Mobile-First Approach

```css
/* Navigation.module.css */

/* Base: mobile styles */
.nav {
  display: flex;
  flex-direction: column;
  position: fixed;
  inset: 0;
  background: var(--color-bg-primary);
  transform: translateX(-100%);
  transition: transform 0.3s ease;
}

.navOpen {
  transform: translateX(0);
}

/* Tablet and up */
@media (min-width: 768px) {
  .nav {
    position: static;
    flex-direction: row;
    transform: none;
    background: transparent;
  }

  .menuButton {
    display: none;
  }
}
```

## Container Queries

Container queries allow styling based on parent container size rather than viewport.

### Setup

```css
/* CardGrid.module.css */
.container {
  container-type: inline-size;
  container-name: card-grid;
}

.card {
  padding: 16px;
}

@container card-grid (min-width: 400px) {
  .card {
    padding: 24px;
    display: flex;
    gap: 16px;
  }
}

@container card-grid (min-width: 600px) {
  .card {
    padding: 32px;
  }

  .cardImage {
    width: 200px;
    flex-shrink: 0;
  }
}
```

```tsx
import styles from './CardGrid.module.css';

function CardGrid({ children }: { children: React.ReactNode }) {
  return (
    <div className={styles.container}>
      {children}
    </div>
  );
}

function Card({ title, image }: { title: string; image: string }) {
  return (
    <article className={styles.card}>
      <img className={styles.cardImage} src={image} alt="" />
      <h3>{title}</h3>
    </article>
  );
}
```

### Container Query Units

```css
.card {
  /* Units relative to container */
  font-size: clamp(14px, 3cqi, 18px); /* cqi = container query inline */
  padding: 5cqw; /* cqw = container query width */
}
```

<Callout type="info" title="Browser Support">
Container queries are supported in all modern browsers. Check caniuse.com for specific version requirements.
</Callout>

## useMediaQuery Hook

### Implementation

```tsx
import { useState, useEffect } from 'react';

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(() => {
    if (typeof window === 'undefined') return false;
    return window.matchMedia(query).matches;
  });

  useEffect(() => {
    const mediaQuery = window.matchMedia(query);
    setMatches(mediaQuery.matches);

    const handler = (event: MediaQueryListEvent) => {
      setMatches(event.matches);
    };

    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, [query]);

  return matches;
}
```

### Usage

```tsx
function Navigation() {
  const isMobile = useMediaQuery('(max-width: 767px)');
  const [menuOpen, setMenuOpen] = useState(false);

  // Close menu when switching to desktop
  useEffect(() => {
    if (!isMobile) {
      setMenuOpen(false);
    }
  }, [isMobile]);

  return (
    <nav>
      {isMobile && (
        <button onClick={() => setMenuOpen(!menuOpen)}>
          Menu
        </button>
      )}
      <ul className={isMobile && !menuOpen ? styles.hidden : undefined}>
        <li>Home</li>
        <li>About</li>
        <li>Contact</li>
      </ul>
    </nav>
  );
}
```

### Typed Breakpoint Hook

```tsx
const breakpoints = {
  sm: '(min-width: 640px)',
  md: '(min-width: 768px)',
  lg: '(min-width: 1024px)',
  xl: '(min-width: 1280px)',
} as const;

type Breakpoint = keyof typeof breakpoints;

export function useBreakpoint(breakpoint: Breakpoint): boolean {
  return useMediaQuery(breakpoints[breakpoint]);
}

// Usage
function Component() {
  const isDesktop = useBreakpoint('lg');
  return isDesktop ? <DesktopView /> : <MobileView />;
}
```

## Responsive Props Pattern

### Component with Responsive Values

```tsx
type ResponsiveValue<T> = T | { sm?: T; md?: T; lg?: T; xl?: T };

interface StackProps {
  gap: ResponsiveValue<number>;
  direction: ResponsiveValue<'row' | 'column'>;
  children: React.ReactNode;
}

function Stack({ gap, direction, children }: StackProps) {
  const resolvedGap = useResponsiveValue(gap);
  const resolvedDirection = useResponsiveValue(direction);

  return (
    <div
      style={{
        display: 'flex',
        flexDirection: resolvedDirection,
        gap: `${resolvedGap}px`,
      }}
    >
      {children}
    </div>
  );
}

// Usage
<Stack gap={{ sm: 8, md: 16, lg: 24 }} direction={{ sm: 'column', md: 'row' }}>
  <Card />
  <Card />
</Stack>
```

### useResponsiveValue Hook

```tsx
function useResponsiveValue<T>(value: ResponsiveValue<T>): T {
  const sm = useMediaQuery('(min-width: 640px)');
  const md = useMediaQuery('(min-width: 768px)');
  const lg = useMediaQuery('(min-width: 1024px)');
  const xl = useMediaQuery('(min-width: 1280px)');

  if (typeof value !== 'object' || value === null) {
    return value as T;
  }

  const responsive = value as { sm?: T; md?: T; lg?: T; xl?: T };

  if (xl && responsive.xl !== undefined) return responsive.xl;
  if (lg && responsive.lg !== undefined) return responsive.lg;
  if (md && responsive.md !== undefined) return responsive.md;
  if (sm && responsive.sm !== undefined) return responsive.sm;

  // Return first defined value as base
  return (responsive.sm ?? responsive.md ?? responsive.lg ?? responsive.xl) as T;
}
```

<Callout type="warn" title="Performance Consideration">
Multiple useMediaQuery hooks cause multiple event listeners. For complex responsive logic, prefer CSS-based solutions.
</Callout>

## Breakpoint Constants

### Shared Configuration

```typescript
// config/breakpoints.ts
export const breakpoints = {
  sm: 640,
  md: 768,
  lg: 1024,
  xl: 1280,
  '2xl': 1536,
} as const;

export type BreakpointKey = keyof typeof breakpoints;

export const mediaQueries = {
  sm: `(min-width: ${breakpoints.sm}px)`,
  md: `(min-width: ${breakpoints.md}px)`,
  lg: `(min-width: ${breakpoints.lg}px)`,
  xl: `(min-width: ${breakpoints.xl}px)`,
  '2xl': `(min-width: ${breakpoints['2xl']}px)`,
} as const;

// For CSS-in-JS or inline styles
export const mq = {
  sm: `@media ${mediaQueries.sm}`,
  md: `@media ${mediaQueries.md}`,
  lg: `@media ${mediaQueries.lg}`,
  xl: `@media ${mediaQueries.xl}`,
  '2xl': `@media ${mediaQueries['2xl']}`,
} as const;
```

### Generating CSS Custom Properties

```typescript
// Generate CSS variables file
function generateBreakpointCSS(): string {
  return `:root {
${Object.entries(breakpoints)
  .map(([key, value]) => `  --breakpoint-${key}: ${value}px;`)
  .join('\n')}
}`;
}
```

## SSR Considerations

### Hydration Mismatch Prevention

```tsx
function useMediaQuery(query: string): boolean | undefined {
  const [matches, setMatches] = useState<boolean | undefined>(undefined);

  useEffect(() => {
    const mediaQuery = window.matchMedia(query);
    setMatches(mediaQuery.matches);

    const handler = (event: MediaQueryListEvent) => {
      setMatches(event.matches);
    };

    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, [query]);

  return matches;
}

// Usage with loading state
function ResponsiveComponent() {
  const isMobile = useMediaQuery('(max-width: 767px)');

  // Render nothing or skeleton during hydration
  if (isMobile === undefined) {
    return <Skeleton />;
  }

  return isMobile ? <MobileView /> : <DesktopView />;
}
```

### CSS-First for Critical Layout

```tsx
// Prefer CSS for initial layout to avoid layout shift
function Layout({ children }: { children: React.ReactNode }) {
  return (
    <div className={styles.layout}>
      {/* Sidebar visibility controlled by CSS */}
      <aside className={styles.sidebar}>
        <Navigation />
      </aside>
      <main className={styles.main}>
        {children}
      </main>
    </div>
  );
}
```

```css
/* Layout.module.css */
.layout {
  display: grid;
  grid-template-columns: 1fr;
}

.sidebar {
  display: none;
}

@media (min-width: 1024px) {
  .layout {
    grid-template-columns: 250px 1fr;
  }

  .sidebar {
    display: block;
  }
}
```

<Callout type="info" title="Best Practice">
Use CSS media queries for layout changes visible on initial render. Use JS hooks only for behavior changes or non-critical UI variations.
</Callout>

---

## When to Use Each Approach

| Approach | Use Case | SSR Safe |
|----------|----------|----------|
| CSS Media Queries | Layout, visibility, spacing | Yes |
| Container Queries | Component-level responsive design | Yes |
| useMediaQuery | Conditional rendering, behavior changes | No* |
| Responsive Props | Flexible component APIs | No* |

*Requires hydration handling

## Common Responsive Patterns

### Responsive Grid

```css
.grid {
  display: grid;
  gap: var(--spacing-md);
  grid-template-columns: 1fr;
}

@media (min-width: 640px) {
  .grid {
    grid-template-columns: repeat(2, 1fr);
  }
}

@media (min-width: 1024px) {
  .grid {
    grid-template-columns: repeat(3, 1fr);
  }
}

/* Or auto-fit for flexible columns */
.gridAuto {
  display: grid;
  gap: var(--spacing-md);
  grid-template-columns: repeat(auto-fit, minmax(280px, 1fr));
}
```

### Responsive Typography

```css
:root {
  --font-size-h1: clamp(1.75rem, 4vw, 3rem);
  --font-size-h2: clamp(1.5rem, 3vw, 2.25rem);
  --font-size-body: clamp(1rem, 1.5vw, 1.125rem);
}

.heading {
  font-size: var(--font-size-h1);
  line-height: 1.2;
}
```

### Show/Hide Elements

```css
.mobileOnly {
  display: block;
}

.desktopOnly {
  display: none;
}

@media (min-width: 768px) {
  .mobileOnly {
    display: none;
  }

  .desktopOnly {
    display: block;
  }
}
```

```tsx
// Component usage
<span className={styles.mobileOnly}>Menu</span>
<span className={styles.desktopOnly}>Navigation</span>
```
