## Overview

Generic components allow you to create reusable components that maintain type safety across different data types. Instead of using `any` or complex union types, generics let the consumer specify the exact type at usage time.

## Basic Generic Components

### Simple List Component

```tsx
interface ListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  keyExtractor: (item: T, index: number) => string | number;
}

function List<T>({ items, renderItem, keyExtractor }: ListProps<T>) {
  return (
    <ul>
      {items.map((item, index) => (
        <li key={keyExtractor(item, index)}>
          {renderItem(item, index)}
        </li>
      ))}
    </ul>
  );
}

// Usage
interface User {
  id: number;
  name: string;
}

const users: User[] = [
  { id: 1, name: 'Alice' },
  { id: 2, name: 'Bob' },
];

<List
  items={users}
  keyExtractor={(user) => user.id}
  renderItem={(user) => <span>{user.name}</span>}
/>
```

<Callout type="info">
TypeScript infers the generic type from the `items` prop. You rarely need to specify it explicitly.
</Callout>

## Generic Select Component

```tsx
interface SelectOption<T> {
  value: T;
  label: string;
}

interface SelectProps<T> {
  options: SelectOption<T>[];
  value: T;
  onChange: (value: T) => void;
  getOptionKey: (option: SelectOption<T>) => string | number;
}

function Select<T>({
  options,
  value,
  onChange,
  getOptionKey,
}: SelectProps<T>) {
  const handleChange = (event: React.ChangeEvent<HTMLSelectElement>) => {
    const selectedIndex = event.target.selectedIndex;
    onChange(options[selectedIndex].value);
  };

  return (
    <select
      value={options.findIndex((opt) => opt.value === value)}
      onChange={handleChange}
    >
      {options.map((option) => (
        <option key={getOptionKey(option)} value={options.indexOf(option)}>
          {option.label}
        </option>
      ))}
    </select>
  );
}

// Usage with string values
<Select
  options={[
    { value: 'draft', label: 'Draft' },
    { value: 'published', label: 'Published' },
  ]}
  value="draft"
  onChange={(status) => console.log(status)} // status is string
  getOptionKey={(opt) => opt.value}
/>

// Usage with object values
interface Status {
  code: number;
  name: string;
}

<Select<Status>
  options={[
    { value: { code: 1, name: 'Active' }, label: 'Active' },
    { value: { code: 2, name: 'Inactive' }, label: 'Inactive' },
  ]}
  value={{ code: 1, name: 'Active' }}
  onChange={(status) => console.log(status.code)} // status is Status
  getOptionKey={(opt) => opt.value.code}
/>
```

## Constrained Generics

### Requiring Specific Properties

```tsx
interface WithId {
  id: string | number;
}

interface DataTableProps<T extends WithId> {
  data: T[];
  columns: Array<{
    key: keyof T;
    header: string;
  }>;
  onRowClick?: (row: T) => void;
}

function DataTable<T extends WithId>({
  data,
  columns,
  onRowClick,
}: DataTableProps<T>) {
  return (
    <table>
      <thead>
        <tr>
          {columns.map((col) => (
            <th key={String(col.key)}>{col.header}</th>
          ))}
        </tr>
      </thead>
      <tbody>
        {data.map((row) => (
          <tr key={row.id} onClick={() => onRowClick?.(row)}>
            {columns.map((col) => (
              <td key={String(col.key)}>{String(row[col.key])}</td>
            ))}
          </tr>
        ))}
      </tbody>
    </table>
  );
}

// Usage - TypeScript ensures items have 'id'
interface Product {
  id: number;
  name: string;
  price: number;
}

<DataTable<Product>
  data={products}
  columns={[
    { key: 'name', header: 'Product Name' },
    { key: 'price', header: 'Price' },
  ]}
  onRowClick={(product) => console.log(product.name)}
/>
```

### Multiple Constraints

```tsx
interface Identifiable {
  id: string | number;
}

interface Timestamped {
  createdAt: Date;
  updatedAt: Date;
}

function AuditableList<T extends Identifiable & Timestamped>({
  items,
}: {
  items: T[];
}) {
  return (
    <ul>
      {items.map((item) => (
        <li key={item.id}>
          Created: {item.createdAt.toLocaleDateString()}
        </li>
      ))}
    </ul>
  );
}
```

## Generic Form Components

### Form Field Component

```tsx
interface FormFieldProps<T, K extends keyof T> {
  form: T;
  field: K;
  label: string;
  onChange: (field: K, value: T[K]) => void;
}

function FormField<T, K extends keyof T>({
  form,
  field,
  label,
  onChange,
}: FormFieldProps<T, K>) {
  const value = form[field];

  const handleChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    onChange(field, event.target.value as T[K]);
  };

  return (
    <div className="form-field">
      <label>{label}</label>
      <input value={String(value)} onChange={handleChange} />
    </div>
  );
}

// Usage
interface UserForm {
  name: string;
  email: string;
  age: number;
}

const [form, setForm] = useState<UserForm>({
  name: '',
  email: '',
  age: 0,
});

<FormField
  form={form}
  field="name"
  label="Name"
  onChange={(field, value) => setForm({ ...form, [field]: value })}
/>
```

## Generic Render Props

```tsx
interface DataLoaderProps<T> {
  url: string;
  children: (data: T, loading: boolean, error: Error | null) => React.ReactNode;
}

function DataLoader<T>({ url, children }: DataLoaderProps<T>) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    setLoading(true);
    fetch(url)
      .then((res) => res.json())
      .then((json: T) => {
        setData(json);
        setLoading(false);
      })
      .catch((err) => {
        setError(err);
        setLoading(false);
      });
  }, [url]);

  return <>{data && children(data, loading, error)}</>;
}

// Usage
interface Post {
  id: number;
  title: string;
  body: string;
}

<DataLoader<Post[]> url="/api/posts">
  {(posts, loading, error) => (
    <ul>
      {posts.map((post) => (
        <li key={post.id}>{post.title}</li>
      ))}
    </ul>
  )}
</DataLoader>
```

## Generic with Default Type

```tsx
interface PaginatedListProps<T = unknown> {
  items: T[];
  pageSize?: number;
  renderItem: (item: T) => React.ReactNode;
}

function PaginatedList<T = unknown>({
  items,
  pageSize = 10,
  renderItem,
}: PaginatedListProps<T>) {
  const [page, setPage] = useState(0);

  const start = page * pageSize;
  const pageItems = items.slice(start, start + pageSize);

  return (
    <div>
      {pageItems.map(renderItem)}
      <div className="pagination">
        <button onClick={() => setPage((p) => Math.max(0, p - 1))}>
          Previous
        </button>
        <button onClick={() => setPage((p) => p + 1)}>
          Next
        </button>
      </div>
    </div>
  );
}
```

## Pros and Cons

### Generic Components

| Pros | Cons |
|------|------|
| Type-safe across all usages | More complex implementation |
| Eliminates `any` | Learning curve for team |
| IDE autocomplete works | Longer type definitions |
| Reusable across data types | Can be over-engineered |
| Compile-time error catching | Inference not always perfect |

### Non-Generic (Union) Approach

| Pros | Cons |
|------|------|
| Simpler to understand | Limited flexibility |
| Explicit about supported types | Grows unwieldy with many types |
| No generic syntax | Repeated code for each type |

## When to Use

| Scenario | Use Generic Components |
|----------|------------------------|
| Data display components (tables, lists) | Yes |
| Form field wrappers | Yes |
| Selection components (dropdown, autocomplete) | Yes |
| Data fetching wrappers | Yes |
| Components used with multiple data types | Yes |

## When NOT to Use

| Scenario | Reason |
|----------|--------|
| Single-purpose components | Unnecessary complexity |
| Components with fixed props | No type variation needed |
| Simple presentational components | Over-engineering |
| Components always used with one type | Use concrete type |

<Callout type="warn">
Don't use generics just because you can. If a component only ever works with one type, use that type directly.
</Callout>

## Generic forwardRef Components

```tsx
interface GenericInputProps<T> {
  value: T;
  onChange: (value: T) => void;
  parse: (input: string) => T;
  format: (value: T) => string;
}

function GenericInputInner<T>(
  { value, onChange, parse, format }: GenericInputProps<T>,
  ref: React.ForwardedRef<HTMLInputElement>
) {
  return (
    <input
      ref={ref}
      value={format(value)}
      onChange={(e) => onChange(parse(e.target.value))}
    />
  );
}

// Type assertion needed for forwardRef with generics
const GenericInput = forwardRef(GenericInputInner) as <T>(
  props: GenericInputProps<T> & { ref?: React.ForwardedRef<HTMLInputElement> }
) => React.ReactElement;

// Usage
<GenericInput<number>
  value={42}
  onChange={(n) => console.log(n)}
  parse={(s) => parseInt(s, 10)}
  format={(n) => String(n)}
/>
```

## Pattern: Generic Context

```tsx
interface StoreContextValue<T> {
  state: T;
  dispatch: React.Dispatch<Action<T>>;
}

type Action<T> = { type: 'SET'; payload: T } | { type: 'RESET' };

function createStore<T>(initialState: T) {
  const Context = React.createContext<StoreContextValue<T> | null>(null);

  function Provider({ children }: { children: React.ReactNode }) {
    const [state, dispatch] = useReducer(
      (state: T, action: Action<T>): T => {
        switch (action.type) {
          case 'SET':
            return action.payload;
          case 'RESET':
            return initialState;
          default:
            return state;
        }
      },
      initialState
    );

    return (
      <Context.Provider value={{ state, dispatch }}>
        {children}
      </Context.Provider>
    );
  }

  function useStore(): StoreContextValue<T> {
    const context = useContext(Context);
    if (!context) {
      throw new Error('useStore must be used within Provider');
    }
    return context;
  }

  return { Provider, useStore };
}

// Usage
interface AppState {
  user: string | null;
  theme: 'light' | 'dark';
}

const { Provider: AppProvider, useStore: useAppStore } = createStore<AppState>({
  user: null,
  theme: 'light',
});
```

## Best Practices

1. **Let TypeScript infer** when possible - avoid explicit type annotations at call site
2. **Use constraints** (`extends`) to ensure required properties exist
3. **Provide sensible defaults** for generic types when applicable
4. **Name type parameters** descriptively for complex generics (`TItem`, `TValue` instead of `T`)
5. **Document the expected shape** with JSDoc when constraints aren't enough
6. **Test with multiple types** to ensure the component is truly generic

 * A virtualized list component for efficient rendering of large datasets.

  items: T[];
  /** Height of each item in pixels */
  itemHeight: number;
  /** Render function for each item */
