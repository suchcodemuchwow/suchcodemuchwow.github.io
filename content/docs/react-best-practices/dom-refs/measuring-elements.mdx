## getBoundingClientRect

Get element dimensions and position relative to viewport:

```tsx
function ElementInfo() {
  const elementRef = useRef<HTMLDivElement>(null);
  const [rect, setRect] = useState<DOMRect | null>(null);

  useLayoutEffect(() => {
    if (elementRef.current) {
      setRect(elementRef.current.getBoundingClientRect());
    }
  }, []);

  return (
    <>
      <div ref={elementRef} style={{ padding: 20, background: '#eee' }}>
        Measure me
      </div>
      {rect && (
        <dl>
          <dt>Width</dt><dd>{rect.width}px</dd>
          <dt>Height</dt><dd>{rect.height}px</dd>
          <dt>Top</dt><dd>{rect.top}px</dd>
          <dt>Left</dt><dd>{rect.left}px</dd>
        </dl>
      )}
    </>
  );
}
```

### DOMRect Properties

| Property | Description |
|----------|-------------|
| `width` | Element width including padding and border |
| `height` | Element height including padding and border |
| `top` | Distance from viewport top to element top |
| `left` | Distance from viewport left to element left |
| `bottom` | Distance from viewport top to element bottom |
| `right` | Distance from viewport left to element right |
| `x` | Same as `left` |
| `y` | Same as `top` |

<Callout type="info" title="Viewport Relative">
`getBoundingClientRect` returns positions relative to the viewport, not the document. Values change as user scrolls.
</Callout>

## ResizeObserver

Detect element size changes:

```tsx
function useElementSize<T extends HTMLElement>() {
  const ref = useRef<T>(null);
  const [size, setSize] = useState({ width: 0, height: 0 });

  useLayoutEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new ResizeObserver((entries) => {
      const entry = entries[0];
      if (entry) {
        const { width, height } = entry.contentRect;
        setSize({ width, height });
      }
    });

    observer.observe(element);

    return () => observer.disconnect();
  }, []);

  return { ref, size };
}

// Usage
function ResponsiveComponent() {
  const { ref, size } = useElementSize<HTMLDivElement>();

  return (
    <div ref={ref} style={{ resize: 'both', overflow: 'auto' }}>
      {size.width > 600 ? <WideLayout /> : <NarrowLayout />}
    </div>
  );
}
```

### ResizeObserver Entry Properties

```tsx
const observer = new ResizeObserver((entries) => {
  for (const entry of entries) {
    // Content box (excludes padding)
    const contentRect = entry.contentRect;

    // Border box (includes padding and border)
    const borderBoxSize = entry.borderBoxSize[0];

    // Content box size
    const contentBoxSize = entry.contentBoxSize[0];

    console.log({
      contentWidth: contentRect.width,
      contentHeight: contentRect.height,
      borderBoxWidth: borderBoxSize?.inlineSize,
      borderBoxHeight: borderBoxSize?.blockSize,
    });
  }
});
```

## IntersectionObserver

Detect when elements enter or leave the viewport:

```tsx
function useIntersection<T extends HTMLElement>(
  options?: IntersectionObserverInit
) {
  const ref = useRef<T>(null);
  const [isIntersecting, setIsIntersecting] = useState(false);
  const [ratio, setRatio] = useState(0);

  useEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        setIsIntersecting(entry.isIntersecting);
        setRatio(entry.intersectionRatio);
      },
      options
    );

    observer.observe(element);

    return () => observer.disconnect();
  }, [options?.root, options?.rootMargin, options?.threshold]);

  return { ref, isIntersecting, ratio };
}

// Usage: Lazy load images
function LazyImage({ src, alt }: { src: string; alt: string }) {
  const { ref, isIntersecting } = useIntersection<HTMLDivElement>({
    rootMargin: '100px', // Load 100px before entering viewport
    threshold: 0,
  });
  const [loaded, setLoaded] = useState(false);

  return (
    <div ref={ref}>
      {(isIntersecting || loaded) && (
        <img src={src} alt={alt} onLoad={() => setLoaded(true)} />
      )}
    </div>
  );
}
```

### Infinite Scroll Pattern

```tsx
function InfiniteList({ loadMore }: { loadMore: () => void }) {
  const sentinelRef = useRef<HTMLDivElement>(null);
  const loadingRef = useRef(false);

  useEffect(() => {
    const sentinel = sentinelRef.current;
    if (!sentinel) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting && !loadingRef.current) {
          loadingRef.current = true;
          loadMore();
          // Reset after load completes
          setTimeout(() => {
            loadingRef.current = false;
          }, 100);
        }
      },
      { rootMargin: '200px' }
    );

    observer.observe(sentinel);

    return () => observer.disconnect();
  }, [loadMore]);

  return <div ref={sentinelRef} aria-hidden="true" />;
}
```

### Visibility Tracking

```tsx
function useVisibilityTracking(elementId: string) {
  const ref = useRef<HTMLElement>(null);
  const trackedRef = useRef(false);

  useEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting && !trackedRef.current) {
          trackedRef.current = true;
          // Track impression
          analytics.track('element_viewed', { elementId });
        }
      },
      { threshold: 0.5 } // 50% visible
    );

    observer.observe(element);

    return () => observer.disconnect();
  }, [elementId]);

  return ref;
}
```

## useLayoutEffect for Measurements

### Prevent Visual Flicker

```tsx
function Tooltip({ children, content }: Props) {
  const triggerRef = useRef<HTMLButtonElement>(null);
  const tooltipRef = useRef<HTMLDivElement>(null);
  const [position, setPosition] = useState({ top: 0, left: 0 });
  const [isVisible, setIsVisible] = useState(false);

  // Calculate position before paint
  useLayoutEffect(() => {
    if (!isVisible) return;
    if (!triggerRef.current || !tooltipRef.current) return;

    const triggerRect = triggerRef.current.getBoundingClientRect();
    const tooltipRect = tooltipRef.current.getBoundingClientRect();

    // Position above trigger, centered
    setPosition({
      top: triggerRect.top - tooltipRect.height - 8,
      left: triggerRect.left + (triggerRect.width - tooltipRect.width) / 2,
    });
  }, [isVisible]);

  return (
    <>
      <button
        ref={triggerRef}
        onMouseEnter={() => setIsVisible(true)}
        onMouseLeave={() => setIsVisible(false)}
      >
        {children}
      </button>
      {isVisible && (
        <div
          ref={tooltipRef}
          style={{
            position: 'fixed',
            top: position.top,
            left: position.left,
          }}
        >
          {content}
        </div>
      )}
    </>
  );
}
```

### Measure and Adjust

```tsx
function AutoHeight({ children }: { children: React.ReactNode }) {
  const contentRef = useRef<HTMLDivElement>(null);
  const [height, setHeight] = useState<number | 'auto'>('auto');

  useLayoutEffect(() => {
    if (contentRef.current) {
      setHeight(contentRef.current.scrollHeight);
    }
  }, [children]);

  return (
    <div style={{ height, overflow: 'hidden', transition: 'height 0.3s' }}>
      <div ref={contentRef}>{children}</div>
    </div>
  );
}
```

<Callout type="warn" title="useLayoutEffect Rule">
Use `useLayoutEffect` when you need to measure and then immediately update based on that measurement. It prevents flicker but runs synchronously, so keep it fast.
</Callout>

## Resize Handling

### Window Resize

```tsx
function useWindowSize() {
  const [size, setSize] = useState({
    width: window.innerWidth,
    height: window.innerHeight,
  });

  useEffect(() => {
    const handleResize = () => {
      setSize({
        width: window.innerWidth,
        height: window.innerHeight,
      });
    };

    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  return size;
}
```

### Debounced Resize

```tsx
function useDebouncedWindowSize(delay: number = 250) {
  const [size, setSize] = useState({
    width: window.innerWidth,
    height: window.innerHeight,
  });

  useEffect(() => {
    let timeoutId: number;

    const handleResize = () => {
      clearTimeout(timeoutId);
      timeoutId = window.setTimeout(() => {
        setSize({
          width: window.innerWidth,
          height: window.innerHeight,
        });
      }, delay);
    };

    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      clearTimeout(timeoutId);
    };
  }, [delay]);

  return size;
}
```

### Container Queries Alternative

```tsx
function useContainerQuery<T extends HTMLElement>(
  breakpoint: number
): { ref: React.RefObject<T>; isWide: boolean } {
  const ref = useRef<T>(null);
  const [isWide, setIsWide] = useState(false);

  useLayoutEffect(() => {
    const element = ref.current;
    if (!element) return;

    const observer = new ResizeObserver(([entry]) => {
      setIsWide(entry.contentRect.width >= breakpoint);
    });

    observer.observe(element);

    return () => observer.disconnect();
  }, [breakpoint]);

  return { ref, isWide };
}

// Usage
function Card() {
  const { ref, isWide } = useContainerQuery<HTMLDivElement>(400);

  return (
    <div ref={ref} className={isWide ? 'card-wide' : 'card-narrow'}>
      {/* Content adapts to container, not viewport */}
    </div>
  );
}
```

## Virtual Scrolling Basics

Render only visible items for large lists:

```tsx
interface VirtualListProps<T> {
  items: T[];
  itemHeight: number;
  containerHeight: number;
  renderItem: (item: T, index: number) => React.ReactNode;
}

function VirtualList<T>({
  items,
  itemHeight,
  containerHeight,
  renderItem,
}: VirtualListProps<T>) {
  const [scrollTop, setScrollTop] = useState(0);

  const totalHeight = items.length * itemHeight;
  const startIndex = Math.floor(scrollTop / itemHeight);
  const endIndex = Math.min(
    startIndex + Math.ceil(containerHeight / itemHeight) + 1,
    items.length
  );

  const visibleItems = items.slice(startIndex, endIndex);
  const offsetY = startIndex * itemHeight;

  const handleScroll = (e: React.UIEvent<HTMLDivElement>) => {
    setScrollTop(e.currentTarget.scrollTop);
  };

  return (
    <div
      style={{ height: containerHeight, overflow: 'auto' }}
      onScroll={handleScroll}
    >
      <div style={{ height: totalHeight, position: 'relative' }}>
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleItems.map((item, i) => (
            <div key={startIndex + i} style={{ height: itemHeight }}>
              {renderItem(item, startIndex + i)}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
```

### Variable Height Items

```tsx
function useVirtualList<T>({
  items,
  estimatedItemHeight,
  containerRef,
}: {
  items: T[];
  estimatedItemHeight: number;
  containerRef: React.RefObject<HTMLElement>;
}) {
  const [scrollTop, setScrollTop] = useState(0);
  const [containerHeight, setContainerHeight] = useState(0);
  const measuredHeights = useRef<Map<number, number>>(new Map());

  // Track actual heights as items render
  const measureItem = useCallback((index: number, height: number) => {
    measuredHeights.current.set(index, height);
  }, []);

  const getItemHeight = useCallback((index: number) => {
    return measuredHeights.current.get(index) ?? estimatedItemHeight;
  }, [estimatedItemHeight]);

  // Calculate which items are visible
  const getVisibleRange = useCallback(() => {
    let start = 0;
    let accumulatedHeight = 0;

    // Find start index
    for (let i = 0; i < items.length; i++) {
      const height = getItemHeight(i);
      if (accumulatedHeight + height > scrollTop) {
        start = i;
        break;
      }
      accumulatedHeight += height;
    }

    // Find end index
    let end = start;
    let visibleHeight = 0;
    for (let i = start; i < items.length; i++) {
      visibleHeight += getItemHeight(i);
      end = i + 1;
      if (visibleHeight > containerHeight) break;
    }

    return { start, end, offsetY: accumulatedHeight };
  }, [items.length, scrollTop, containerHeight, getItemHeight]);

  return { getVisibleRange, measureItem, setScrollTop, setContainerHeight };
}
```

## Performance Considerations

### Batch Measurements

```tsx
// Bad: Multiple reflows
function measureAll(elements: HTMLElement[]) {
  return elements.map(el => {
    el.style.width = '100px'; // Write
    return el.offsetHeight; // Read - forces reflow
  });
}

// Good: Batch reads and writes
function measureAllBatched(elements: HTMLElement[]) {
  // All writes first
  elements.forEach(el => {
    el.style.width = '100px';
  });

  // Then all reads
  return elements.map(el => el.offsetHeight);
}
```

### Throttle Scroll Handlers

```tsx
function useThrottledScroll(callback: () => void, delay: number) {
  const lastRun = useRef(0);
  const timeoutRef = useRef<number>();

  useEffect(() => {
    const handleScroll = () => {
      const now = Date.now();

      if (now - lastRun.current >= delay) {
        callback();
        lastRun.current = now;
      } else {
        clearTimeout(timeoutRef.current);
        timeoutRef.current = window.setTimeout(() => {
          callback();
          lastRun.current = Date.now();
        }, delay - (now - lastRun.current));
      }
    };

    window.addEventListener('scroll', handleScroll, { passive: true });

    return () => {
      window.removeEventListener('scroll', handleScroll);
      clearTimeout(timeoutRef.current);
    };
  }, [callback, delay]);
}
```

### Use CSS When Possible

| JavaScript Approach | CSS Alternative |
|---------------------|-----------------|
| Window resize listener | CSS media queries |
| ResizeObserver for layout | CSS container queries |
| IntersectionObserver for animations | `animation-timeline: view()` |
| Manual scroll position | `scroll-snap` |

<Callout type="info" title="Performance Tip">
Prefer CSS solutions over JavaScript measurements when possible. CSS is optimized by the browser and won't cause JavaScript-induced reflows.
</Callout>

### Avoid Layout Thrashing

```tsx
// Properties that trigger layout recalculation when read:
// - offsetTop, offsetLeft, offsetWidth, offsetHeight
// - scrollTop, scrollLeft, scrollWidth, scrollHeight
// - clientTop, clientLeft, clientWidth, clientHeight
// - getComputedStyle()
// - getBoundingClientRect()

// Cache measurements
function useStableMeasurement<T extends HTMLElement>() {
  const ref = useRef<T>(null);
  const cachedRect = useRef<DOMRect | null>(null);

  const measure = useCallback(() => {
    if (ref.current) {
      cachedRect.current = ref.current.getBoundingClientRect();
    }
    return cachedRect.current;
  }, []);

  const getCached = useCallback(() => cachedRect.current, []);

  return { ref, measure, getCached };
}
```
