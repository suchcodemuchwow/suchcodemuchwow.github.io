## Why Keyboard Navigation Matters

Many users rely on keyboards due to motor disabilities, visual impairments, or personal preference. All interactive elements must be keyboard accessible.

## Common Key Patterns

Standard keyboard conventions users expect:

| Key | Action |
|-----|--------|
| Tab | Move to next focusable element |
| Shift+Tab | Move to previous focusable element |
| Enter | Activate link or button |
| Space | Activate button, toggle checkbox |
| Escape | Close/dismiss |
| Arrow keys | Navigate within widgets |
| Home/End | Jump to first/last item |

## onKeyDown Handlers

Use `onKeyDown` for keyboard interactions. Avoid `onKeyUp` for activation.

### Basic Key Handler

```tsx
function InteractiveCard() {
  const handleKeyDown = (e: React.KeyboardEvent<HTMLDivElement>) => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      console.log('Activated');
    }
  };

  return (
    <div
      role="button"
      tabIndex={0}
      onClick={() => console.log('Clicked')}
      onKeyDown={handleKeyDown}
    >
      Press Enter or Space
    </div>
  );
}
```

<Callout type="info">
Native `<button>` elements handle Enter and Space automatically. Only add key handlers for custom interactive elements.
</Callout>

### Key Event Properties

| Property | Description |
|----------|-------------|
| `e.key` | Key value ('Enter', 'Escape', 'a') |
| `e.code` | Physical key ('KeyA', 'Space') |
| `e.shiftKey` | Shift held |
| `e.ctrlKey` | Ctrl held |
| `e.altKey` | Alt held |
| `e.metaKey` | Cmd (Mac) / Win held |

```tsx
function ShortcutHandler() {
  const handleKeyDown = (e: React.KeyboardEvent) => {
    // Ctrl+S or Cmd+S
    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
      e.preventDefault();
      console.log('Save');
    }
  };

  return (
    <div onKeyDown={handleKeyDown}>
      Press Ctrl+S to save
    </div>
  );
}
```

## Keyboard Shortcuts

Implement shortcuts carefully to avoid conflicts with browser and screen reader shortcuts.

### Safe Shortcut Patterns

| Pattern | Example | Notes |
|---------|---------|-------|
| Modifier + Letter | Ctrl+S | May conflict with browser |
| Single letter (when focused) | 'j' for next | Only in app-specific contexts |
| Function keys | F2 to edit | Often safe |

### Global Shortcut Hook

```tsx
import { useEffect, useCallback } from 'react';

function useGlobalShortcut(
  key: string,
  callback: () => void,
  modifiers: { ctrl?: boolean; shift?: boolean; alt?: boolean } = {}
) {
  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    const ctrlMatch = modifiers.ctrl ? (e.ctrlKey || e.metaKey) : true;
    const shiftMatch = modifiers.shift ? e.shiftKey : !e.shiftKey;
    const altMatch = modifiers.alt ? e.altKey : !e.altKey;

    if (e.key === key && ctrlMatch && shiftMatch && altMatch) {
      e.preventDefault();
      callback();
    }
  }, [key, callback, modifiers]);

  useEffect(() => {
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown]);
}

// Usage
function App() {
  useGlobalShortcut('k', () => openSearch(), { ctrl: true });
  useGlobalShortcut('Escape', () => closeModal());

  return <div>...</div>;
}
```

<Callout type="warn">
Avoid single-key shortcuts globally. Screen reader users use letter keys for navigation commands.
</Callout>

## Arrow Key Navigation

Use arrow keys for navigation within composite widgets.

### List Navigation

```tsx
import { useState, useRef, useCallback } from 'react';

interface ListItem {
  id: string;
  label: string;
}

function NavigableList({ items }: { items: ListItem[] }) {
  const [activeIndex, setActiveIndex] = useState(0);
  const listRef = useRef<HTMLUListElement>(null);
  const itemRefs = useRef<HTMLLIElement[]>([]);

  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
    let newIndex = activeIndex;

    switch (e.key) {
      case 'ArrowDown':
        newIndex = Math.min(activeIndex + 1, items.length - 1);
        break;
      case 'ArrowUp':
        newIndex = Math.max(activeIndex - 1, 0);
        break;
      case 'Home':
        newIndex = 0;
        break;
      case 'End':
        newIndex = items.length - 1;
        break;
      default:
        return;
    }

    e.preventDefault();
    setActiveIndex(newIndex);
    itemRefs.current[newIndex]?.focus();
  }, [activeIndex, items.length]);

  return (
    <ul
      ref={listRef}
      role="listbox"
      aria-label="Options"
      onKeyDown={handleKeyDown}
    >
      {items.map((item, index) => (
        <li
          key={item.id}
          ref={(el) => {
            if (el) itemRefs.current[index] = el;
          }}
          role="option"
          tabIndex={index === activeIndex ? 0 : -1}
          aria-selected={index === activeIndex}
        >
          {item.label}
        </li>
      ))}
    </ul>
  );
}
```

### Typeahead Search

```tsx
import { useState, useRef, useCallback, useEffect } from 'react';

interface ListItem {
  id: string;
  label: string;
}

function TypeaheadList({ items }: { items: ListItem[] }) {
  const [activeIndex, setActiveIndex] = useState(0);
  const [searchString, setSearchString] = useState('');
  const searchTimeout = useRef<number>();
  const itemRefs = useRef<HTMLLIElement[]>([]);

  // Clear search string after delay
  useEffect(() => {
    if (searchString) {
      searchTimeout.current = window.setTimeout(() => {
        setSearchString('');
      }, 500);
    }
    return () => clearTimeout(searchTimeout.current);
  }, [searchString]);

  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
    // Handle arrow keys
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
      e.preventDefault();
      const newIndex = e.key === 'ArrowDown'
        ? Math.min(activeIndex + 1, items.length - 1)
        : Math.max(activeIndex - 1, 0);
      setActiveIndex(newIndex);
      itemRefs.current[newIndex]?.focus();
      return;
    }

    // Typeahead for printable characters
    if (e.key.length === 1 && !e.ctrlKey && !e.metaKey) {
      const newSearch = searchString + e.key.toLowerCase();
      setSearchString(newSearch);

      const matchIndex = items.findIndex((item) =>
        item.label.toLowerCase().startsWith(newSearch)
      );

      if (matchIndex !== -1) {
        setActiveIndex(matchIndex);
        itemRefs.current[matchIndex]?.focus();
      }
    }
  }, [activeIndex, items, searchString]);

  return (
    <ul role="listbox" onKeyDown={handleKeyDown}>
      {items.map((item, index) => (
        <li
          key={item.id}
          ref={(el) => {
            if (el) itemRefs.current[index] = el;
          }}
          role="option"
          tabIndex={index === activeIndex ? 0 : -1}
          aria-selected={index === activeIndex}
        >
          {item.label}
        </li>
      ))}
    </ul>
  );
}
```

## Enter/Space for Activation

Different elements have different activation keys.

| Element | Enter | Space |
|---------|-------|-------|
| Button | Activates | Activates |
| Link | Activates | Scrolls page |
| Checkbox | Activates | Activates |
| Radio | Activates | Activates |
| Select option | Selects | Selects |

```tsx
// Custom toggle - needs both Enter and Space
function CustomToggle() {
  const [on, setOn] = useState(false);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      setOn(!on);
    }
  };

  return (
    <div
      role="switch"
      aria-checked={on}
      tabIndex={0}
      onClick={() => setOn(!on)}
      onKeyDown={handleKeyDown}
    >
      {on ? 'On' : 'Off'}
    </div>
  );
}
```

<Callout type="info">
Call `e.preventDefault()` for Space to prevent page scrolling.
</Callout>

## Escape for Dismiss

Escape should close modals, dropdowns, and overlays.

```tsx
import { useEffect, useCallback } from 'react';

function useEscapeKey(onEscape: () => void, enabled: boolean = true) {
  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    if (e.key === 'Escape') {
      onEscape();
    }
  }, [onEscape]);

  useEffect(() => {
    if (enabled) {
      document.addEventListener('keydown', handleKeyDown);
      return () => document.removeEventListener('keydown', handleKeyDown);
    }
  }, [enabled, handleKeyDown]);
}

// Usage
function Modal({ open, onClose }: { open: boolean; onClose: () => void }) {
  useEscapeKey(onClose, open);

  if (!open) return null;

  return (
    <div role="dialog" aria-modal="true">
      <button onClick={onClose}>Close</button>
    </div>
  );
}
```

### Nested Escapable Elements

```tsx
function NestedModals() {
  const [outerOpen, setOuterOpen] = useState(false);
  const [innerOpen, setInnerOpen] = useState(false);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Escape') {
      e.stopPropagation(); // Prevent outer from closing
      if (innerOpen) {
        setInnerOpen(false);
      } else {
        setOuterOpen(false);
      }
    }
  };

  return (
    <div onKeyDown={handleKeyDown}>
      {outerOpen && (
        <div role="dialog">
          Outer modal
          <button onClick={() => setInnerOpen(true)}>
            Open inner
          </button>

          {innerOpen && (
            <div role="dialog">
              Inner modal
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```

## Focus Within Pattern

Track focus within a container for styling or behavior.

### Using :focus-within CSS

```tsx
function SearchBox() {
  return (
    <div className="search-container">
      <input type="search" placeholder="Search..." />
      <button type="submit">Go</button>
    </div>
  );
}
```

```css
.search-container {
  border: 1px solid #ccc;
}

.search-container:focus-within {
  border-color: #005fcc;
  box-shadow: 0 0 0 2px rgba(0, 95, 204, 0.2);
}
```

### JavaScript Focus Within

```tsx
import { useState, useCallback } from 'react';

function FocusWithinContainer({ children }: { children: React.ReactNode }) {
  const [hasFocus, setHasFocus] = useState(false);

  const handleFocus = useCallback(() => setHasFocus(true), []);
  const handleBlur = useCallback((e: React.FocusEvent<HTMLDivElement>) => {
    // Check if focus moved outside container
    if (!e.currentTarget.contains(e.relatedTarget)) {
      setHasFocus(false);
    }
  }, []);

  return (
    <div
      onFocus={handleFocus}
      onBlur={handleBlur}
      data-focus-within={hasFocus}
    >
      {children}
    </div>
  );
}
```

### Dropdown with Focus Within

```tsx
import { useState, useCallback, useRef } from 'react';

function Dropdown() {
  const [open, setOpen] = useState(false);
  const containerRef = useRef<HTMLDivElement>(null);

  const handleBlur = useCallback((e: React.FocusEvent) => {
    // Close when focus leaves dropdown
    if (!containerRef.current?.contains(e.relatedTarget)) {
      setOpen(false);
    }
  }, []);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Escape') {
      setOpen(false);
    }
  };

  return (
    <div
      ref={containerRef}
      onBlur={handleBlur}
      onKeyDown={handleKeyDown}
    >
      <button
        type="button"
        aria-expanded={open}
        aria-haspopup="menu"
        onClick={() => setOpen(!open)}
      >
        Menu
      </button>

      {open && (
        <ul role="menu">
          <li role="menuitem" tabIndex={-1}>Option 1</li>
          <li role="menuitem" tabIndex={-1}>Option 2</li>
          <li role="menuitem" tabIndex={-1}>Option 3</li>
        </ul>
      )}
    </div>
  );
}
```

## Complete Widget Example

Accessible combobox with keyboard navigation.

```tsx
import { useState, useRef, useCallback, useId } from 'react';

interface Option {
  value: string;
  label: string;
}

interface ComboboxProps {
  options: Option[];
  placeholder?: string;
  onChange: (value: string) => void;
}

function Combobox({ options, placeholder, onChange }: ComboboxProps) {
  const [open, setOpen] = useState(false);
  const [inputValue, setInputValue] = useState('');
  const [activeIndex, setActiveIndex] = useState(-1);

  const inputRef = useRef<HTMLInputElement>(null);
  const listRef = useRef<HTMLUListElement>(null);
  const optionRefs = useRef<HTMLLIElement[]>([]);

  const listboxId = useId();

  const filteredOptions = options.filter((opt) =>
    opt.label.toLowerCase().includes(inputValue.toLowerCase())
  );

  const handleInputKeyDown = useCallback((e: React.KeyboardEvent) => {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        if (!open) {
          setOpen(true);
          setActiveIndex(0);
        } else {
          setActiveIndex((prev) =>
            Math.min(prev + 1, filteredOptions.length - 1)
          );
        }
        break;

      case 'ArrowUp':
        e.preventDefault();
        setActiveIndex((prev) => Math.max(prev - 1, 0));
        break;

      case 'Enter':
        e.preventDefault();
        if (activeIndex >= 0 && filteredOptions[activeIndex]) {
          selectOption(filteredOptions[activeIndex]);
        }
        break;

      case 'Escape':
        setOpen(false);
        setActiveIndex(-1);
        break;

      case 'Home':
        if (open) {
          e.preventDefault();
          setActiveIndex(0);
        }
        break;

      case 'End':
        if (open) {
          e.preventDefault();
          setActiveIndex(filteredOptions.length - 1);
        }
        break;
    }
  }, [open, activeIndex, filteredOptions]);

  const selectOption = (option: Option) => {
    setInputValue(option.label);
    onChange(option.value);
    setOpen(false);
    setActiveIndex(-1);
    inputRef.current?.focus();
  };

  const handleBlur = (e: React.FocusEvent) => {
    if (!e.currentTarget.contains(e.relatedTarget)) {
      setOpen(false);
    }
  };

  return (
    <div onBlur={handleBlur}>
      <input
        ref={inputRef}
        type="text"
        role="combobox"
        aria-expanded={open}
        aria-controls={listboxId}
        aria-activedescendant={
          activeIndex >= 0 ? `option-${activeIndex}` : undefined
        }
        aria-autocomplete="list"
        placeholder={placeholder}
        value={inputValue}
        onChange={(e) => {
          setInputValue(e.target.value);
          setOpen(true);
          setActiveIndex(-1);
        }}
        onKeyDown={handleInputKeyDown}
        onFocus={() => setOpen(true)}
      />

      {open && filteredOptions.length > 0 && (
        <ul
          ref={listRef}
          id={listboxId}
          role="listbox"
        >
          {filteredOptions.map((option, index) => (
            <li
              key={option.value}
              ref={(el) => {
                if (el) optionRefs.current[index] = el;
              }}
              id={`option-${index}`}
              role="option"
              aria-selected={index === activeIndex}
              onClick={() => selectOption(option)}
            >
              {option.label}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

## Quick Reference

| Pattern | Keys | Implementation |
|---------|------|----------------|
| Activate button | Enter, Space | Native or `onKeyDown` |
| Navigate list | Arrow Up/Down | `onKeyDown` + roving tabIndex |
| Close overlay | Escape | `onKeyDown` or global listener |
| Navigate tabs | Arrow Left/Right | `onKeyDown` + roving tabIndex |
| Select option | Enter | `onKeyDown` handler |
| Type to search | Letters | Typeahead pattern |
| Jump to start | Home | `onKeyDown` handler |
| Jump to end | End | `onKeyDown` handler |
