## Context for State Management

Context solves prop drilling but isn't a general state manager:

| Use Context For | Don't Use Context For |
|-----------------|----------------------|
| Theme, locale, auth | Frequently changing data |
| Deep prop drilling (3+ levels) | High-frequency updates |
| Data many components need | Data few components need |
| App-wide configuration | Local component state |

```tsx
// Good context use case: auth state
const AuthContext = createContext<AuthContextValue | null>(null);

interface AuthContextValue {
  user: User | null;
  login: (credentials: Credentials) => Promise<void>;
  logout: () => void;
}

function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);

  const login = async (credentials: Credentials) => {
    const user = await authApi.login(credentials);
    setUser(user);
  };

  const logout = () => {
    authApi.logout();
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider');
  }
  return context;
}
```

## Multiple Contexts Strategy

Split contexts by domain and update frequency:

```tsx
// Separate contexts by concern
const UserContext = createContext<User | null>(null);
const ThemeContext = createContext<Theme>('light');
const LocaleContext = createContext<Locale>('en');
const NotificationsContext = createContext<Notification[]>([]);

function AppProviders({ children }: { children: ReactNode }) {
  return (
    <UserProvider>
      <ThemeProvider>
        <LocaleProvider>
          <NotificationsProvider>
            {children}
          </NotificationsProvider>
        </LocaleProvider>
      </ThemeProvider>
    </UserProvider>
  );
}
```

### Why Split?

| Single Context | Multiple Contexts |
|----------------|-------------------|
| Any change re-renders all consumers | Only affected consumers re-render |
| Hard to find what triggered update | Clear update source |
| Difficult to test in isolation | Easy to mock individual contexts |
| All-or-nothing consumption | Consume only what you need |

```tsx
// Bad: mega-context
interface AppState {
  user: User | null;
  theme: Theme;
  locale: Locale;
  notifications: Notification[];
  sidebarOpen: boolean;
  // ... more
}

// Changing theme re-renders everything that uses AppContext!
const AppContext = createContext<AppState>(/* ... */);

// Good: split by domain
// Changing theme only re-renders theme consumers
function ThemeToggle() {
  const theme = useContext(ThemeContext); // Only subscribes to theme
  // ...
}
```

## Context + useReducer

Combine for complex state logic:

```tsx
type State = {
  items: CartItem[];
  isOpen: boolean;
};

type Action =
  | { type: 'ADD_ITEM'; item: CartItem }
  | { type: 'REMOVE_ITEM'; id: string }
  | { type: 'UPDATE_QUANTITY'; id: string; quantity: number }
  | { type: 'CLEAR' }
  | { type: 'TOGGLE_OPEN' };

function cartReducer(state: State, action: Action): State {
  switch (action.type) {
    case 'ADD_ITEM': {
      const existing = state.items.find(i => i.id === action.item.id);
      if (existing) {
        return {
          ...state,
          items: state.items.map(i =>
            i.id === action.item.id
              ? { ...i, quantity: i.quantity + action.item.quantity }
              : i
          ),
        };
      }
      return { ...state, items: [...state.items, action.item] };
    }
    case 'REMOVE_ITEM':
      return { ...state, items: state.items.filter(i => i.id !== action.id) };
    case 'UPDATE_QUANTITY':
      return {
        ...state,
        items: state.items.map(i =>
          i.id === action.id ? { ...i, quantity: action.quantity } : i
        ),
      };
    case 'CLEAR':
      return { ...state, items: [] };
    case 'TOGGLE_OPEN':
      return { ...state, isOpen: !state.isOpen };
  }
}

const CartContext = createContext<{
  state: State;
  dispatch: Dispatch<Action>;
} | null>(null);

function CartProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(cartReducer, { items: [], isOpen: false });

  return (
    <CartContext.Provider value={{ state, dispatch }}>
      {children}
    </CartContext.Provider>
  );
}
```

<Callout type="info" title="Benefits">
useReducer centralizes state logic, making it testable outside React and easier to debug.
</Callout>

## Splitting Read/Write Contexts

Prevent unnecessary re-renders by separating state and dispatch:

```tsx
// Split into two contexts
const CartStateContext = createContext<State | null>(null);
const CartDispatchContext = createContext<Dispatch<Action> | null>(null);

function CartProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(cartReducer, initialState);

  return (
    <CartStateContext.Provider value={state}>
      <CartDispatchContext.Provider value={dispatch}>
        {children}
      </CartDispatchContext.Provider>
    </CartStateContext.Provider>
  );
}

// Custom hooks for each context
function useCartState() {
  const state = useContext(CartStateContext);
  if (!state) throw new Error('useCartState must be used within CartProvider');
  return state;
}

function useCartDispatch() {
  const dispatch = useContext(CartDispatchContext);
  if (!dispatch) throw new Error('useCartDispatch must be used within CartProvider');
  return dispatch;
}

// Or combine when you need both
function useCart() {
  return { state: useCartState(), dispatch: useCartDispatch() };
}
```

### Why Split?

```tsx
// Component only needs to dispatch - never re-renders on state change
function AddToCartButton({ product }: { product: Product }) {
  const dispatch = useCartDispatch(); // Stable reference, no re-renders

  return (
    <button onClick={() => dispatch({ type: 'ADD_ITEM', item: { ...product, quantity: 1 } })}>
      Add to Cart
    </button>
  );
}

// Component only needs state - doesn't need dispatch
function CartCount() {
  const { items } = useCartState();
  return <span>{items.reduce((sum, i) => sum + i.quantity, 0)}</span>;
}
```

| Pattern | Re-renders When |
|---------|-----------------|
| Single context | State OR dispatch changes (state changes = all consumers re-render) |
| Split contexts | Only when the specific context value changes |

## Context Selector Patterns

React Context doesn't support selectors natively, but you can optimize:

### Pattern 1: Multiple Granular Contexts

```tsx
// Instead of one context with all user data
const UserNameContext = createContext<string>('');
const UserEmailContext = createContext<string>('');
const UserRoleContext = createContext<Role>('guest');

function UserProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);

  return (
    <UserNameContext.Provider value={user?.name ?? ''}>
      <UserEmailContext.Provider value={user?.email ?? ''}>
        <UserRoleContext.Provider value={user?.role ?? 'guest'}>
          {children}
        </UserRoleContext.Provider>
      </UserEmailContext.Provider>
    </UserNameContext.Provider>
  );
}

// Component only re-renders when name changes
function Greeting() {
  const name = useContext(UserNameContext);
  return <h1>Hello, {name}</h1>;
}
```

### Pattern 2: Memoized Component Wrapper

```tsx
const DataContext = createContext<{ items: Item[]; selectedId: string | null }>({
  items: [],
  selectedId: null,
});

// Wrapper that only re-renders when selected item changes
function SelectedItemDisplay() {
  const { items, selectedId } = useContext(DataContext);
  const selectedItem = items.find(i => i.id === selectedId);

  return <MemoizedDisplay item={selectedItem} />;
}

const MemoizedDisplay = memo(function Display({ item }: { item: Item | undefined }) {
  if (!item) return null;
  return <div>{item.name}</div>;
});
```

### Pattern 3: Subscription-Based Context

```tsx
type Listener<T> = (state: T) => void;

function createStore<T>(initialState: T) {
  let state = initialState;
  const listeners = new Set<Listener<T>>();

  return {
    getState: () => state,
    setState: (newState: T) => {
      state = newState;
      listeners.forEach(listener => listener(state));
    },
    subscribe: (listener: Listener<T>) => {
      listeners.add(listener);
      return () => listeners.delete(listener);
    },
  };
}

type Store<T> = ReturnType<typeof createStore<T>>;

const StoreContext = createContext<Store<AppState> | null>(null);

// Hook that only re-renders when selected value changes
function useSelector<T, S>(selector: (state: T) => S): S {
  const store = useContext(StoreContext) as Store<T>;
  const [selectedState, setSelectedState] = useState(() => selector(store.getState()));

  useEffect(() => {
    const unsubscribe = store.subscribe(state => {
      const newSelected = selector(state);
      setSelectedState(prev => {
        // Only update if selected value changed
        return Object.is(prev, newSelected) ? prev : newSelected;
      });
    });
    return unsubscribe;
  }, [store, selector]);

  return selectedState;
}

// Usage - only re-renders when item count changes
function ItemCount() {
  const count = useSelector((state: AppState) => state.items.length);
  return <span>{count} items</span>;
}
```

<Callout type="warn" title="Complexity Tradeoff">
The subscription pattern adds complexity. Consider if you truly need fine-grained updates before implementing.
</Callout>

## Context Best Practices

| Practice | Reason |
|----------|--------|
| Create custom hooks (`useAuth`, `useCart`) | Better DX, can throw if missing provider |
| Split by domain | Limits re-render scope |
| Split read/write | Components that only dispatch never re-render |
| Use `useMemo` for object values | Prevents unnecessary re-renders |
| Place providers high, consumers low | Minimize provider nesting |

```tsx
// Always memoize object/array context values
function DataProvider({ children }: { children: ReactNode }) {
  const [items, setItems] = useState<Item[]>([]);

  // Without useMemo, new object every render = all consumers re-render
  const value = useMemo(
    () => ({ items, setItems }),
    [items]
  );

  return (
    <DataContext.Provider value={value}>
      {children}
    </DataContext.Provider>
  );
}
```
